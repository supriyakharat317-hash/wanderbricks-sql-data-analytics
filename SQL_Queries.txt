


select
  (select count(*) from `databricks_wanderbricks_dataset_dais_2025`.`wanderbricks`.`users`) as total_users,
  (select count(*) from `databricks_wanderbricks_dataset_dais_2025`.`wanderbricks`.`properties`) as total_properties,
  (select count(*) from `databricks_wanderbricks_dataset_dais_2025`.`wanderbricks`.`bookings`) as total_bookings,
  (select count(*) from `databricks_wanderbricks_dataset_dais_2025`.`wanderbricks`.`hosts`) as total_hosts,
  (select count(*) from `databricks_wanderbricks_dataset_dais_2025`.`wanderbricks`.`cities`) as total_cities,
  (select count(*) from `databricks_wanderbricks_dataset_dais_2025`.`wanderbricks`.`amenities`) as total_amenities,
  (select count(*) from `databricks_wanderbricks_dataset_dais_2025`.`wanderbricks`.`employees`) as total_employees;

--check booking status

select status, count(*) as total_bookings
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings
group by status;

--total revenue

select round(sum(total_amount),2) as total_revenue
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings
where status = 'confirmed';

--avg booking value

select round(avg(total_amount),2) as total_revenue
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings
where status = 'confirmed';

-- what percentage of total booking are cancelled?
--cancelled booking / total booking * 100


with cancelled as (
    select count(*) as cancelled_count
    from  databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings
where status = 'cancelled'
),
total as (
    select count(*) as total_count
    from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings
)
select round(cancelled_count / total_count * 100,2) as cancelled_percent
from cancelled, total;

--avg property price

select round(avg(base_price),2) as avg_property_price
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.properties;

-- top 5 cities by properties count

select c.city, count(*) as total_properties
from  databricks_wanderbricks_dataset_dais_2025.wanderbricks.properties p
join  databricks_wanderbricks_dataset_dais_2025.wanderbricks.cities c
on p.city_id = c.city_id
group by c.city
order by total_properties desc
limit 5;

--top 5 cities by revenue

select c.city,round(sum(total_amount),2) as total_revenue
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings b
join databricks_wanderbricks_dataset_dais_2025.wanderbricks.properties p
on b.property_id = p.property_id
join databricks_wanderbricks_dataset_dais_2025.wanderbricks.cities c
on p.city_id = c.city_id
group by c.city
order by total_revenue desc
limit 5;

--top 5 hosts by property count

select host_id, count(*) as total_properties
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.properties
group by host_id
order by total_properties desc
limit 5;

--booking by property type

select p.property_type, count(*) as total_bookings
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings b
join databricks_wanderbricks_dataset_dais_2025.wanderbricks.properties p
on b.property_id = p.property_id
group by p.property_type
order by total_bookings desc;

--monthly booking trend

select year(created_at) as year,
month(created_at) as month,
count(*) as total_bookings
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings
group by year, month
order by year, month;

--booking cancellation by city

select c.city, count(*) as total_cancellations
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.bookings b
join databricks_wanderbricks_dataset_dais_2025.wanderbricks.properties p
on b.property_id = p.property_id
join databricks_wanderbricks_dataset_dais_2025.wanderbricks.cities c
on p.city_id = c.city_id
where b.status = 'cancelled'
group by c.city
order by total_cancellations desc
limit 5;


--most popular amenities

select a.name, count(*) as total_amenities
from databricks_wanderbricks_dataset_dais_2025.wanderbricks.amenities a
join databricks_wanderbricks_dataset_dais_2025.wanderbricks.property_amenities p
  on a.amenity_id = p.amenity_id
group by a.name
order by total_amenities desc
limit 5;
